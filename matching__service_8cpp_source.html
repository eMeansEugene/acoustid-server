14MatchOutput MatchingService::Match(const std::vector<uint8_t>& bytes) {
15 // 1. Декодировать фрагмент.
16 const audio::AudioData audio = decoder_.DecodeFromBytes(bytes);
17
18 // 2. DSP-пайплайн: сэмплы → спектрограмма → пики → fingerprints.
19 core::FingerprintResult fp_result = engine_.Process(audio.samples_);
20
21 // 3. Собрать уникальные хэши для запроса в БД.
22 // Без дедупликации: если хэш H встречается во фрагменте 3 раза,
23 // FindMatches вернёт результаты 3 раза → BuildHashMatches раздует
24 // число совпадений квадратично → шум заглушает сигнал.
25 std::unordered_set<uint32_t> unique_set;
26 for (const core::Fingerprint& fp : fp_result.fingerprints) {
27 unique_set.insert(fp.hash_);
28 }
29 std::vector<uint32_t> hashes(unique_set.begin(), unique_set.end());
30
31 // 4. Найти совпадения в БД.
32 const std::vector<HashLookupResult> lookup_results = repository_.FindMatches(hashes);
33
34 // 5. Соединить результаты БД с данными фрагмента.
35 const std::vector<core::HashMatch> matches = BuildHashMatches(fp_result.fingerprints, lookup_results);
36
37 // 6. Голосование.
38 const std::optional<core::MatchResult> match = voter_.Vote(matches);
39
40 // 7. Диагностика.

42 diag.sample_rate = audio.sample_rate_;
43 diag.duration_sec = audio.duration_sec_;

45 diag.num_peaks = fp_result.peaks.size();
46 diag.num_fingerprints = fp_result.fingerprints.size();
47 diag.num_unique_hashes = hashes.size();
48 diag.num_db_matches = lookup_results.size();
49 diag.num_hash_matches = matches.size();
50
51 return MatchOutput{
52 .fingerprint_result = std::move(fp_result),
53 .match_result = match,
54 .diagnostics = diag,

58std::vector<core::HashMatch> MatchingService::BuildHashMatches(
59 const std::vector<core::Fingerprint>& fragment_fps,
60 const std::vector<HashLookupResult>& lookup_results) {
61 // Построить индекс: hash → список anchor_frame во фрагменте.
62 // Один хэш может встречаться в нескольких позициях фрагмента.
63 std::unordered_map<uint32_t, std::vector<std::size_t>> hash_to_fragment_frames;
64 for (const core::Fingerprint& fp : fragment_fps) {
65 hash_to_fragment_frames[fp.hash_].push_back(fp.anchor_frame_);
66 }
67
68 // Для каждого совпадения из БД — найти все позиции этого хэша во фрагменте
69 // и создать HashMatch для каждой пары (трек-позиция, фрагмент-позиция).
70 std::vector<core::HashMatch> matches;
71 for (const HashLookupResult& lr : lookup_results) {
72 const auto it = hash_to_fragment_frames.find(lr.hash_);
73 if (it == hash_to_fragment_frames.end()) {
74 continue; // не должно случиться, но на всякий случай
75 }
76 for (const std::size_t fragment_frame : it->second) {
77 matches.push_back(core::HashMatch{
78 .track_id_ = lr.track_id_,
79 .track_anchor_frame_ = lr.track_anchor_frame_,
80 .fragment_anchor_frame_ = fragment_frame,
81 });
82 }
83 }
84
85 return matches;
86}
